cmake_minimum_required(VERSION 3.22)
project(boltrt VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BOLTRT_G1 "Build the experimental Unitree G1 backend" OFF)
option(BOLTRT_RP1 "Build the experimental GPL-3.0 RP1 backend" OFF)
option(BOLTRT_VIEWER "Build the native MuJoCo viewer" ON)
set(MUJOCO_ROOT "${PROJECT_SOURCE_DIR}/third_party/mujoco-3.10.0" CACHE PATH "MuJoCo distribution")
if(CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
  set(ORT_ARCH aarch64)
else()
  set(ORT_ARCH x64)
endif()
set(ORT_ROOT "${PROJECT_SOURCE_DIR}/third_party/onnxruntime-linux-${ORT_ARCH}-1.23.2" CACHE PATH "ONNX Runtime distribution")
set(UNITREE_SDK_ROOT "${PROJECT_SOURCE_DIR}/third_party/unitree_sdk2-official" CACHE PATH "Official Unitree SDK2")
set(RP1_SDK_ROOT "${PROJECT_SOURCE_DIR}/third_party/roboparty_deploy-official" CACHE PATH "Official RoboParty SDK")
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
find_path(MUJOCO_INCLUDE mujoco/mujoco.h HINTS "${MUJOCO_ROOT}/include" REQUIRED)
find_library(MUJOCO_LIB NAMES mujoco HINTS "${MUJOCO_ROOT}/lib" REQUIRED)
find_path(ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${ORT_ROOT}/include" REQUIRED)
find_library(ORT_LIB onnxruntime HINTS "${ORT_ROOT}/lib" REQUIRED)
find_library(ZMQ_LIB NAMES zmq libzmq.so.5 REQUIRED)

add_library(boltrt_core
  src/config/config.cpp src/config/hf_hub.cpp src/backend/backend.cpp
  src/observation/observation.cpp src/observation/common.cpp src/runtime/task.cpp src/runtime/policy_loop.cpp src/runtime/terminal.cpp)
target_include_directories(boltrt_core PUBLIC include third_party/include)
target_link_libraries(boltrt_core PUBLIC Eigen3::Eigen yaml-cpp Threads::Threads)
target_compile_options(boltrt_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(boltrt_mujoco_util src/utils/mujoco/model.cpp)
target_include_directories(boltrt_mujoco_util PUBLIC "${MUJOCO_INCLUDE}")
target_link_libraries(boltrt_mujoco_util PUBLIC boltrt_core "${MUJOCO_LIB}")
add_library(boltrt_ort src/utils/onnx_inference.cpp)
target_include_directories(boltrt_ort PRIVATE "${ORT_INCLUDE}" third_party/include)
target_link_libraries(boltrt_ort PUBLIC boltrt_core "${ORT_LIB}")
add_subdirectory(src/tasks/track)
add_subdirectory(src/backend)

add_executable(boltrt apps/boltrt.cpp)
target_link_libraries(boltrt PRIVATE boltrt_backend boltrt_track boltrt_ort)
target_compile_definitions(boltrt PRIVATE BOLTRT_CONFIG_DIR="${PROJECT_SOURCE_DIR}/configs")
foreach(robot G1 RP1)
  if(BOLTRT_${robot})
    string(TOLOWER "${robot}" robot_target)
    target_compile_definitions(boltrt PRIVATE BOLTRT_${robot})
    target_link_libraries(boltrt PRIVATE boltrt_${robot_target})
  endif()
endforeach()
install(TARGETS boltrt RUNTIME DESTINATION bin)
install(DIRECTORY configs/ DESTINATION share/boltrt/configs)

include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE boltrt_track_observation)
  add_test(NAME core_contracts COMMAND core_tests)
  add_test(NAME cli_help COMMAND boltrt --help)
  add_test(NAME cli_tasks COMMAND boltrt --list-tasks)
  set(fixture "${PROJECT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME base_loop_cli COMMAND boltrt --robot "${fixture}/base_robot.yaml"
    --policy "${fixture}/base_policy.yaml" --backend "${fixture}/base_backend.yaml"
    --start --steps 10 --no-keyboard)
  set_tests_properties(base_loop_cli PROPERTIES PASS_REGULAR_EXPRESSION "ticks=10")
  add_test(NAME tracking_loop_cli COMMAND boltrt --task tracking --robot "${fixture}/base_robot.yaml"
    --policy "${fixture}/base_policy.yaml" --backend "${fixture}/base_backend.yaml"
    --motion "${fixture}/motion/c_float32.npz" --start --steps 2 --no-keyboard)
  # Record ten ticks, then replay them: every recorded state must reach the policy.
  set(replay_states "${CMAKE_CURRENT_BINARY_DIR}/replay_states.jsonl")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/replay_backend.yaml" "type: replay\npath: ${replay_states}\n")
  add_test(NAME replay_record COMMAND boltrt --robot "${fixture}/base_robot.yaml"
    --policy "${fixture}/base_policy.yaml" --backend "${fixture}/base_backend.yaml"
    --start --steps 10 --no-keyboard --record "${replay_states}")
  set_tests_properties(replay_record PROPERTIES FIXTURES_SETUP replay_states)
  add_test(NAME replay_roundtrip COMMAND boltrt --robot "${fixture}/base_robot.yaml"
    --policy "${fixture}/base_policy.yaml" --backend "${CMAKE_CURRENT_BINARY_DIR}/replay_backend.yaml"
    --headless --start --no-keyboard)
  set_tests_properties(replay_roundtrip PROPERTIES FIXTURES_REQUIRED replay_states
    PASS_REGULAR_EXPRESSION "ticks=10 ")

  add_executable(hardware_tests tests/hardware_tests.cpp)
  target_link_libraries(hardware_tests PRIVATE boltrt_backend)
  add_test(NAME inline_hardware_contracts COMMAND hardware_tests)
  add_executable(inference_tests tests/inference_tests.cpp)
  target_link_libraries(inference_tests PRIVATE boltrt_ort)
  add_test(NAME typed_recurrent_onnx COMMAND inference_tests "${PROJECT_SOURCE_DIR}/tests/fixtures/recurrent.onnx")
  add_executable(motion_tests tests/motion_tests.cpp)
  target_link_libraries(motion_tests PRIVATE boltrt_track_motion)
  add_test(NAME npz_motion_contracts COMMAND motion_tests "${PROJECT_SOURCE_DIR}/tests/fixtures/motion")
  add_executable(external_tests tests/external_tests.cpp)
  target_include_directories(external_tests PRIVATE src/tasks/track/motion third_party/include)
  target_link_libraries(external_tests PRIVATE boltrt_track_observation)
  add_test(NAME external_policy_contracts COMMAND external_tests)
  add_executable(simulation_backend_tests tests/simulation_backend_tests.cpp)
  target_link_libraries(simulation_backend_tests PRIVATE boltrt_backend)
  add_test(NAME simulation_backend_contracts COMMAND simulation_backend_tests "${PROJECT_SOURCE_DIR}/tests/fixtures/motion/robot.xml")
  add_executable(loop_independence_tests tests/loop_independence_tests.cpp)
  target_link_libraries(loop_independence_tests PRIVATE boltrt_track)
  add_test(NAME loop_independence COMMAND loop_independence_tests)
endif()

add_executable(observation_example EXCLUDE_FROM_ALL examples/custom_observation.cpp)
target_link_libraries(observation_example PRIVATE boltrt_core)
